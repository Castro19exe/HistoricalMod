tradition_byzantine_succession = {
	category = regional

	layers = {
		0 = intrigue
		1 = mediterranean
		4 = crown.dds
	}

	is_shown = {
		OR = {
			has_cultural_pillar = heritage_byzantine
			scope:character = {
				has_title = title:e_byzantium
			}
		}
		#replaced by new Byzantine traditions
		has_ep3_dlc_trigger = no
	}
	can_pick = {
		has_cultural_pillar = heritage_byzantine
		scope:character = {
			has_title = title:e_byzantium
		}
		custom_tooltip = {
			text = cannot_have_tradition_court_eunuchs
			NOT = { culture_tradition:tradition_court_eunuchs = { is_in_list = traits } }
		}
		custom_tooltip = {
			text = cannot_have_tradition_merciful_blindings
			NOT = { culture_tradition:tradition_merciful_blindings = { is_in_list = traits } }
		}
	}

	parameters = {
		can_blind_prisoners = yes
		can_castrate_prisoners = yes
		eunuch_trait_bonuses = yes
		can_appoint_chief_eunuch = yes
		physical_disfigurement_blocks_inheritance = yes # Blocks claimant factions, claim wars, and reduce score in elective and acclamation. Also includes infertility for tooltip reasons.
		children_can_be_born_in_the_purple = yes
		disloyal_trait_more_common = yes
		byzantine_faction_differences = yes
		#byzantine_faction_preffered_claimants = yes
	}
	character_modifier = {
		eligible_child_opinion = 20
		short_reign_duration_mult = 0.5
		hostile_scheme_phase_duration_add = medium_scheme_phase_duration_bonus_value
		ai_amenity_target_baseline = 0.75
		ai_amenity_spending = 1
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_courtly = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_courtly_or_communal_desc
				}
			}
			if = {
				limit = {
					NOT = {
						scope:character = {
							has_title = title:c_byzantion
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = byzantion_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_byzantine_succession }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}

			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = { value = 1000 }
}

tradition_imperial_might = {
	category = realm

	layers = {
		0 = martial
		1 = mena\mena1.dds
		2 = rec5 copy 2.dds
		3 = 6.dds
		4 = roman_eagle.dds
	}

	is_shown = {
		OR = {
			this = culture:roman
			scope:character = {
				has_title = title:h_roman_empire
			}
		}
	}
	can_pick = {
		NOR = { 
			has_cultural_tradition = tradition_roman_legacy
			has_cultural_tradition = tradition_stalwart_defenders
		}

		NOT = { # Can not be picked by tribals
			scope:character = {
				has_government = tribal_government
			}
		}
	}
	
	parameters = {
		can_use_conquest_cbs = yes
		cheaper_cbs = yes
	}
	
	province_modifier = {
		monthly_county_control_growth_add = 0.3
	}
	
	character_modifier = {
		men_at_arms_limit = 20
		levy_size = -0.75
		levy_reinforcement_rate = -0.75
		vassal_levy_contribution_mult = -0.75
		#men_at_arms_recruitment_cost = -0.15
		men_at_arms_maintenance = 0.20
		men_at_arms_cap = 10
	}
	
	cost = {
		prestige = {
			add = {
				value = 5000
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
                        culture_pillar:ethos_bellicose = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_desc
				}
			}
			if = {
				limit = {
					NOT = {
						any_culture_county = {
							count >= 90
						}
					}
				}
				add = {
					#value = tradition_unfulfilled_criteria_penalty
					value = 5000
					desc = culture_in_many_counties_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = {
		value = 100
	}
}
